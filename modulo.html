function cekKongruensi() { let a = document.getElementById("bilanganA").value; let b = document.getElementById("bilanganB").value; let n = document.getElementById("modN").value; let output = document.getElementById("hasilKongruensi"); if (a.trim() === "" || b.trim() === "" || n.trim() === "" || Number(n) <= 0) { output.className = "hasil"; output.innerHTML = "Masukkan dua bilangan dan modulus positif!"; return; } a = Number(a); b = Number(b); n = Number(n); let sisaA = ((a % n) + n) % n; let sisaB = ((b % n) + n) % n; if (sisaA === sisaB) { output.className = "hasil kongruen"; output.innerHTML = `

✅ BENAR

${a} ≡ ${b} (mod ${n})

Langkah Perhitungan:

1. ${a} mod ${n} = ${sisaA}

2. ${b} mod ${n} = ${sisaB}

Penjelasan:

Kedua bilangan memiliki sisa pembagian yang sama, yaitu ${sisaA}. Jadi, pernyataan kongruensi tersebut BENAR.

`; } else { output.className = "hasil tidak"; output.innerHTML = `

❌ SALAH

${a} ≢ ${b} (mod ${n})

Langkah Perhitungan:

1. ${a} mod ${n} = ${sisaA}

2. ${b} mod ${n} = ${sisaB}

Penjelasan:

Sisa pembagian ${a} adalah ${sisaA}, sedangkan sisa pembagian ${b} adalah ${sisaB}. Karena sisanya berbeda, pernyataan kongruensi tersebut SALAH.

`; } }
